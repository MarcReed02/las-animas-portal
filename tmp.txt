"use client";

import Link from "next/link";
import { useMemo, useState } from "react";
import { FaEye, FaEyeSlash } from "react-icons/fa6";
import { TbFileUpload } from "react-icons/tb";
import punycode from "punycode/";

type PasswordCheck = {
  score: number;
  message: string | null;
};

const phoneOptions = [
  { iso: "CL", code: "+56" },
  { iso: "AR", code: "+54" },
  { iso: "PE", code: "+51" },
  { iso: "MX", code: "+52" },
  { iso: "US", code: "+1" },
  { iso: "ES", code: "+34" },
  { iso: "CO", code: "+57" },
  { iso: "BR", code: "+55" },
];

const bannedPasswordRanges = [
  [0x0000, 0x001f],
  [0x007f, 0x009f],
  [0x200b, 0x200f],
  [0x2028, 0x202f],
  [0xfeff, 0xfeff],
  [0xd800, 0xdfff],
];

const MIN_PASSWORD = 12;

export default function Page() {
  const [name, setName] = useState("");
  const [phoneIso, setPhoneIso] = useState(phoneOptions[0].iso);
  const [phoneCode, setPhoneCode] = useState(phoneOptions[0].code);
  const [phone, setPhone] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [file, setFile] = useState<File | null>(null);
  const [progress, setProgress] = useState<number | null>(null);
  const [status, setStatus] = useState<null | "ok" | "err">(null);
  const [loading, setLoading] = useState(false);
  const [showPassword, setShowPassword] = useState(false);
  const [errors, setErrors] = useState<Record<string, string>>({});

  const passwordCheck: PasswordCheck = useMemo(() => evaluatePassword(password), [password]);

  const handleUpload = (selected: File | null) => {
    if (!selected) return;
    setFile(selected);
    setProgress(20);
    setTimeout(() => setProgress(70), 400);
    setTimeout(() => setProgress(100), 1200);
  };

  const onSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    const validation = validateForm({ name, phone, email, password, file, passwordCheck });
    setErrors(validation);
    if (Object.keys(validation).length > 0) return;

    setLoading(true);
    setStatus(null);
    try {
      const normalizedName = normalizeName(name);
      const normalizedEmail = normalizeEmail(email);
      const normalizedPassword = password;
      const nationalNumber = phone.replace(/\D/g, "");
      const e164Phone = `${phoneCode}${nationalNumber}`;
      const payload = {
        name: normalizedName,
        email: normalizedEmail,
        password: normalizedPassword,
        phone: e164Phone,
        phoneISO: phoneIso,
        fileName: file?.name,
      };
      console.log("Submitting", payload);
      await new Promise((resolve) => setTimeout(resolve, 1200));
      setStatus("ok");
      setName("");
      setPhone("");
      setEmail("");
      setPassword("");
      setFile(null);
      setProgress(null);
    } catch {
      setStatus("err");
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-[calc(100svh-4rem)] w-full bg-black flex items-center justify-center px-4 py-10">
      <div className="w-full max-w-xl">
        <form
          onSubmit={onSubmit}
          className="space-y-6 rounded-[32px] bg-[#111111] px-8 py-10 text-white shadow-2xl"
        >
          <div className="text-center space-y-1">
            <h1 className="text-3xl md:text-4xl font-extrabold tracking-tight">Create Account</h1>
            <p className="text-white/70 text-sm md:text-base">
              Fill your information below or register with your social account
            </p>
          </div>

          <div className="space-y-5">
            <FieldWrapper label="Name" error={errors.name}>
              <input
                type="text"
                required
                autoComplete="name"
                value={name}
                onChange={(e) => setName(normalizeName(e.target.value, false))}
                onBlur={(e) => setName(normalizeName(e.target.value))}
                placeholder="John Doe"
                aria-invalid={Boolean(errors.name)}
                className={inputClass(errors.name)}
              />
            </FieldWrapper>

            <FieldWrapper label="Phone Number" error={errors.phone}>
              <div className={`flex rounded-2xl overflow-hidden ${errors.phone ? "border border-[#c21e1e]" : "bg-[#0c0c0c] border border-white/15"}`}>
                <div className="relative">
                  <select
                    value={phoneIso}
                    onChange={(e) => {
                      const iso = e.target.value;
                      const option = phoneOptions.find((opt) => opt.iso === iso);
                      if (option) {
                        setPhoneIso(option.iso);
                        setPhoneCode(option.code);
                      }
                    }}
                    className="bg-transparent px-3 py-3 text-sm pr-8 focus:outline-none border-r border-white/10 appearance-none"
                    style={{
                      backgroundImage:
                        "linear-gradient(45deg, transparent 50%, #c21e1e 50%), linear-gradient(135deg, #c21e1e 50%, transparent 50%)",
                      backgroundPosition:
                        "calc(100% - 16px) calc(1em + 2px), calc(100% - 11px) calc(1em + 2px)",
                      backgroundSize: "5px 5px, 5px 5px",
                      backgroundRepeat: "no-repeat",
                    }}
                  >
                    {phoneOptions.map((opt) => (
                      <option key={opt.iso} value={opt.iso}>
                        {opt.code}
                      </option>
                    ))}
                  </select>
                </div>
                <input
                  type="tel"
                  inputMode="tel"
                  autoComplete="tel"
                  placeholder="Enter phone number"
                  value={phone}
                  onChange={(e) => setPhone(e.target.value)}
                  aria-invalid={Boolean(errors.phone)}
                  className="flex-1 bg-transparent px-4 py-3 text-sm placeholder-white/40 focus:outline-none"
                />
              </div>
            </FieldWrapper>

            <FieldWrapper label="Email" error={errors.email}>
              <input
                type="email"
                required
                autoComplete="email"
                placeholder="example@gmail.com"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                aria-invalid={Boolean(errors.email)}
                className={inputClass(errors.email)}
              />
            </FieldWrapper>

            <FieldWrapper label="Password" error={errors.password}>
              <div className="relative">
                <input
                  type={showPassword ? "text" : "password"}
                  required
                  autoComplete="new-password"
                  placeholder="••••••••••"
                  value={password}
                  onChange={(e) => setPassword(e.target.value)}
                  aria-invalid={Boolean(errors.password)}
                  className={inputClass(errors.password)}
                />
                <button
                  type="button"
                  onClick={() => setShowPassword((prev) => !prev)}
                  className="absolute inset-y-0 right-4 flex items-center text-white/60 hover:text-white focus:outline-none"
                >
                  {showPassword ? <FaEyeSlash /> : <FaEye />}
                </button>
              </div>
              <p className="text-xs text-white/50">
                {passwordCheck.message || `Strength score: ${passwordCheck.score}/4`}
              </p>
            </FieldWrapper>

            <FieldWrapper label="Document" error={errors.file}>
            <div className={`rounded-3xl border-2 border-dashed ${errors.file ? "border-[#c21e1e]" : "border-[#c21e1e]/80"} bg-[#0c0c0c] px-4 py-5 space-y-3`}>
              {file ? (
                <div className="space-y-2">
                  <div className="flex items-center justify-between text-sm">
                    <span className="font-semibold">{file.name}</span>
                    <span>{progress ?? 0}%</span>
                  </div>
                  <div className="h-2 rounded-full bg-white/10 overflow-hidden">
                    <div
                      className="h-full bg-[#c21e1e]/80 transition-all"
                      style={{ width: `${progress ?? 0}%` }}
                    />
                  </div>
                </div>
              ) : (
                <div className="text-sm text-white/70">
                  Drag & drop files here or upload file
                  <div className="text-xs text-white/40">PDF, JPG, PNG up to 10MB</div>
                </div>
              )}
              <label className="inline-flex gap-2 items-center rounded-2xl bg-[#c21e1e] hover:bg-[#a41717] px-4 py-2 text-sm font-semibold cursor-pointer transition-colors">
                <TbFileUpload className="h-5 w-5" />
                Choose file
                <input
                  type="file"
                  accept=".pdf,.png,.jpg,.jpeg"
                  className="hidden"
                  onChange={(e) => handleUpload(e.target.files?.[0] ?? null)}
                />
              </label>
            </div>
            </FieldWrapper>
          </div>

          {status === "ok" && (
            <p className="text-green-400 text-center text-sm">Account created successfully</p>
          )}
          {status === "err" && (
            <p className="text-[#c21e1e] text-center text-sm">Something went wrong. Try again.</p>
          )}

          <button
            type="submit"
            disabled={loading}
            className="w-full rounded-2xl bg-[#c21e1e] hover:bg-[#a41717] py-3 text-lg font-semibold transition-colors disabled:opacity-40 disabled:cursor-not-allowed"
          >
            {loading ? "Sending..." : "Sign Up"}
          </button>

          <div className="text-center text-white/70 text-sm">
            Already have an account?{" "}
            <Link href="/auth/login/ingresar" className="text-white hover:underline">
              Log in here
            </Link>
          </div>
        </form>
      </div>
    </div>
  );
}

function normalizeName(value: string, finalize = true) {
  const nfkc = value.normalize("NFKC");
  let cleaned = nfkc.replace(/[^\p{L}\p{M}\s'.-]/gu, "");
  cleaned = cleaned.replace(/\s+/g, " ");
  return finalize ? cleaned.trim().slice(0, 80) : cleaned.slice(0, 80);
}

function normalizeEmail(value: string) {
  const trimmed = value.trim();
  const [local, domainRaw = ""] = trimmed.split("@");
  const domain = domainRaw.toLowerCase();
  const asciiDomain = domain ? punycode.toASCII(domain) : "";
  return `${local}${domain ? "@" + asciiDomain : ""}`.slice(0, 254);
}

function validateForm({
  name,
  phone,
  email,
  password,
  file,
  passwordCheck,
}: {
  name: string;
  phone: string;
  email: string;
  password: string;
  file: File | null;
  passwordCheck: PasswordCheck;
}) {
  const errors: Record<string, string> = {};
  const normalizedName = normalizeName(name);
  if (normalizedName.length < 2 || normalizedName.length > 80) {
    errors.name = "The name must be between 2 and 80 characters.";
  }
  if (!/^[\p{L}\p{M}'.\s-]+$/u.test(normalizedName)) {
    errors.name = "Only letters, spaces, apostrophes, dash and dot are allowed.";
  }

  const digits = phone.replace(/\D/g, "");
  if (!digits) {
    errors.phone = "Phone number is required.";
  }

  const normalizedEmail = normalizeEmail(email);
  if (!normalizedEmail.includes("@")) {
    errors.email = "Email must be valid.";
  }

  if (!password || password.length < MIN_PASSWORD) {
    errors.password = `Password must be at least ${MIN_PASSWORD} characters.`;
  }
  if (passwordCheck.score < 3) {
    errors.password = passwordCheck.message || "Password is too weak.";
  }

  if (!file) {
    errors.file = "Please upload a document.";
  }

  return errors;
}

function evaluatePassword(password: string): PasswordCheck {
  if (!password) {
    return { score: 0, message: "Password is required." };
  }
  if (/^\s|\s$/.test(password)) {
    return { score: 0, message: "No spaces at the beginning or end." };
  }
  for (const ch of password) {
    const code = ch.codePointAt(0)!;
    if (bannedPasswordRanges.some(([start, end]) => code >= start && code <= end)) {
      return { score: 0, message: "Password contains unsupported characters." };
    }
  }
  if (password.length < MIN_PASSWORD) {
    return { score: 1, message: `Must be at least ${MIN_PASSWORD} characters.` };
  }

  let score = 1;
  if (password.length >= 16) score++;
  if (password.length >= 20) score++;

  const variety = [
    /[a-z]/.test(password),
    /[A-Z]/.test(password),
    /[0-9]/.test(password),
    /[^a-zA-Z0-9\s]/.test(password),
  ].filter(Boolean).length;
  if (variety >= 3) score++;

  const commonPatterns = ["password", "123456", "qwerty", "admin", "welcome"];
  if (commonPatterns.some((pat) => password.toLowerCase().includes(pat))) {
    return { score: 1, message: "Password is too common." };
  }

  if (score < 3) {
    return { score, message: "Password needs to be stronger." };
  }
  return { score: Math.min(score, 4), message: null };
}

function inputClass(hasError?: string) {
  return `w-full rounded-2xl px-4 py-3 placeholder-white/40 focus:outline-none focus:ring-2 ${
    hasError
      ? "bg-[#160808] border border-[#c21e1e] focus:ring-[#c21e1e]"
      : "bg-[#0c0c0c] border border-white/15 focus:ring-white/25"
  }`;
}

function FieldWrapper({
  label,
  error,
  children,
}: {
  label: string;
  error?: string;
  children: React.ReactNode;
}) {
  return (
    <div className="space-y-2">
      <label className="text-white/80 text-sm font-semibold">{label}</label>
      {children}
      {error && (
        <div className="flex items-center gap-2 text-sm text-[#c21e1e] bg-[#160808] rounded-2xl border border-[#c21e1e] px-3 py-2">
          <span className="text-lg leading-none">!</span>
          <span>{error}</span>
        </div>
      )}
    </div>
  );
}
